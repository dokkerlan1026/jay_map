# 台北門市地圖 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 建立並發佈一個可查看 5 家台北示範店、啟動單店導航及規劃全店單向最佳路線的響應式靜態網站。

**Architecture:** `dist/` 直接作為靜態部署目錄。純函式集中在 `route-planner.js`，以 Node 內建測試驗證；`app.js` 負責 Leaflet 地圖、清單同步、定位、OSRM 請求與畫面狀態，HTML/CSS 僅負責結構及視覺。

**Tech Stack:** HTML5、CSS、原生 JavaScript ES modules、Leaflet、OpenStreetMap、OSRM HTTP API、Node.js built-in test runner、Sites 靜態部署。

`dist/route-planner.js` 實作：

```js
const OSRM_BASE = "https://router.project-osrm.org";

export function distanceMeters(a, b) {
  const rad = (value) => value * Math.PI / 180;
  const dLat = rad(b.lat - a.lat);
  const dLng = rad(b.lng - a.lng);
  const lat1 = rad(a.lat);
  const lat2 = rad(b.lat);
  const h = Math.sin(dLat / 2) ** 2 + Math.cos(lat1) * Math.cos(lat2) * Math.sin(dLng / 2) ** 2;
  return 6371000 * 2 * Math.atan2(Math.sqrt(h), Math.sqrt(1 - h));
}

export function createEffectiveMatrix(matrix, points) {
  let usedFallback = false;
  const effective = points.map((from, i) => points.map((to, j) => {
    const value = matrix?.[i]?.[j];
    if (Number.isFinite(value)) return value;
    usedFallback = true;
    return distanceMeters(from, to);
  }));
  return { matrix: effective, usedFallback };
}

export function findOptimalOrder(matrix) {
  const destinations = Array.from({ length: matrix.length - 1 }, (_, i) => i + 1);
  let bestOrder = null;
  let bestDistance = Infinity;
  const visit = (prefix, remaining, total) => {
    if (total >= bestDistance) return;
    if (!remaining.length) {
      bestOrder = [0, ...prefix];
      bestDistance = total;
      return;
    }
    const current = prefix.length ? prefix[prefix.length - 1] : 0;
    for (const next of remaining) {
      visit([...prefix, next], remaining.filter((value) => value !== next), total + matrix[current][next]);
    }
  };
  visit([], destinations, 0);
  return bestOrder;
}

const coordinates = (points) => points.map(({ lng, lat }) => `${lng},${lat}`).join(";");
export const buildOsrmTableUrl = (points) => `${OSRM_BASE}/table/v1/driving/${coordinates(points)}?annotations=distance`;
export const buildOsrmRouteUrl = (points) => `${OSRM_BASE}/route/v1/driving/${coordinates(points)}?overview=full&geometries=geojson&steps=false`;
export function buildGoogleMapsUrl(destination, origin) {
  const params = new URLSearchParams({ api: "1", destination: `${destination.lat},${destination.lng}`, travelmode: "driving" });
  if (origin) params.set("origin", `${origin.lat},${origin.lng}`);
  return `https://www.google.com/maps/dir/?${params}`;
}
```
